This notebook attaches the requested revenue classification to the analytical Census and expanded allocation-link tables. It then describes the composition of statutory allocation links by revenue class. The database is a de jure legal inventory; the counts and shares below do not measure revenue amounts, collections, transfers, disbursements, or expenditure execution.

6. Revenue-class summary

Missing Census matches and unmatched instrument_nature values remain visible as Missing. No link is removed from either denominator.

total_links_all <- nrow(atomic_links_all)
total_links_earmarked <- nrow(atomic_links_earmarked)

if (total_links_all == 0L || total_links_earmarked == 0L) {
  stop("Both the full and filtered samples must contain at least one link.")
}

revenue_class_counts_all <- atomic_links_all %>%
  count(revenue_class = revenue_class_display, name = "n_links_all")

revenue_class_counts_earmarked <- atomic_links_earmarked %>%
  count(
    revenue_class = revenue_class_display,
    name = "n_links_earmarked"
  )

revenue_class_summary <- full_join(
  revenue_class_counts_all,
  revenue_class_counts_earmarked,
  by = "revenue_class"
) %>%
  mutate(
    n_links_all = replace_na(n_links_all, 0L),
    n_links_earmarked = replace_na(n_links_earmarked, 0L),
    share_links_all = n_links_all / total_links_all,
    share_links_earmarked = n_links_earmarked / total_links_earmarked
  ) %>%
  select(
    revenue_class,
    n_links_all,
    share_links_all,
    n_links_earmarked,
    share_links_earmarked
  ) %>%
  arrange(desc(n_links_all), revenue_class)

share_and_total_checks <- tibble(
  sample = c(
    "All statutory allocations",
    "Partial and exclusive earmarks"
  ),
  total_atomic_links = c(total_links_all, total_links_earmarked),
  sum_class_link_counts = c(
    sum(revenue_class_summary$n_links_all),
    sum(revenue_class_summary$n_links_earmarked)
  ),
  sum_class_shares = c(
    sum(revenue_class_summary$share_links_all),
    sum(revenue_class_summary$share_links_earmarked)
  ),
  passed = c(
    sum(revenue_class_summary$n_links_all) == total_links_all &&

        sum(revenue_class_summary$share_links_all),

        tolerance = 1e-12

        sum(revenue_class_summary$share_links_earmarked),
        1,
        tolerance = 1e-12

  )
)

if (!all(share_and_total_checks$passed)) {

7. Exact plot data

The two CSV tables below contain exactly the rows and values displayed in their matching figures. The readable labels are derived only for presentation; the stored revenue_class codes remain unchanged.

make_revenue_class_label <- function(values) {
  values %>%

same_table_values <- function(saved, expected) {
  identical(names(saved), names(expected)) &&
    nrow(saved) == nrow(expected) &&
    isTRUE(all.equal(
      as.data.frame(saved),
      as.data.frame(expected),
      check.attributes = FALSE,
      tolerance = 1e-12
    ))
}

commit_output_bundle <- function(stage_paths, output_paths) {
  existed_before <- file.exists(output_paths)
  backup_dir <- tempfile(pattern = "08_revenue_class_backup_")
  dir.create(backup_dir, recursive = TRUE, showWarnings = FALSE)
  backup_paths <- file.path(backup_dir, basename(output_paths))
  names(backup_paths) <- names(output_paths)
  committed <- FALSE

  on.exit({
    if (!committed) {
      for (name in names(output_paths)) {
        if (existed_before[[name]] && file.exists(backup_paths[[name]])) {
          file.copy(
            backup_paths[[name]],
            output_paths[[name]],
            overwrite = TRUE
          )
        } else if (!existed_before[[name]] &&
                   file.exists(output_paths[[name]])) {
          file.remove(output_paths[[name]])
        }
      }
    }
    unlink(stage_dir, recursive = TRUE, force = TRUE)
    unlink(backup_dir, recursive = TRUE, force = TRUE)
  }, add = TRUE)

  for (name in names(output_paths)[existed_before]) {
    if (!file.copy(
      output_paths[[name]],
      backup_paths[[name]],
      overwrite = FALSE
    )) {
      stop("Could not create a rollback copy for ", name, ".")
    }
  }

  commit_ok <- vapply(
    names(output_paths),
    function(name) file.copy(
      stage_paths[[name]],
      output_paths[[name]],
      overwrite = TRUE
    ),
    logical(1)
  )
  if (!all(commit_ok)) {
    stop("Could not commit every requested output file.")
  }

  final_census <- read_all_character(output_paths[["census"]])
  final_expanded <- read_all_character(output_paths[["expanded"]])

  final_preservation_checks <- bind_rows(
    make_preservation_checks(
      "SSA_CENSUS",
      SSA_CENSUS_before,
      final_census
    ),
    make_preservation_checks(
      "SSA_EARMARK_ALLOCATIONS_EXPANDED",
      SSA_EARMARK_ALLOCATIONS_EXPANDED_before,
      final_expanded
    )
  )

  final_analysis_checks <- tibble(
    output = c(
      "08_revenue_class_summary.csv",
      "08_revenue_class_all.csv",
      "08_revenue_class_partial_exclusive.csv",
      "08_revenue_class_all.png",
      "08_revenue_class_partial_exclusive.png"
    ),
    passed = c(
      same_table_values(
        read_analysis_csv(output_paths[["summary"]]),
        revenue_class_summary
      ),
      same_table_values(
        read_analysis_csv(output_paths[["all_plot_data"]]),
        revenue_class_all_plot_data
      ),
      same_table_values(
        read_analysis_csv(output_paths[["earmarked_plot_data"]]),
        revenue_class_earmarked_plot_data
      ),
      file.info(output_paths[["all_figure"]])$size > 10000,
      file.info(output_paths[["earmarked_figure"]])$size > 10000
    )
  )

  if (!all(final_preservation_checks$passed) ||
      !all(final_analysis_checks$passed)) {
    print(final_preservation_checks %>% filter(!passed))
    print(final_analysis_checks %>% filter(!passed))
    stop("Final read-back checks failed; restoring prior files.")
  }

  committed <- TRUE
  list(
    census = final_census,
    expanded = final_expanded,
    preservation_checks = final_preservation_checks,
    analysis_checks = final_analysis_checks
  )
}
